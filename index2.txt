            entry.target.classList.add("visible");
            revealObserver.unobserve(entry.target);
          }
        });

    function clearBagTimers() {
      bagTimers.forEach((timer) => clearTimeout(timer));

    }

    function openBag() {
      clearBagTimers();

      bagOpen = true;
      bagToggle.classList.add("open", "shaking");
      bagToggle.setAttribute("aria-pressed", "true");
      bagHint.textContent = "↑ tap to close";
      bagState.textContent = "Unpacking… here’s what I carry ✨";

      setTimeout(() => {
        bagToggle.classList.remove("shaking");
      }, 500);

      bagItems.forEach((item, index) => {
        const timer = setTimeout(() => {
          item.classList.add("out");
        }, 380 + index * 190);

        bagTimers.push(timer);
      });

      const finalTimer = setTimeout(() => {
        bagState.textContent =
          "That’s my toolkit. Click the bag again to pack it back 👜";
      }, 380 + bagItems.length * 190);

      bagTimers.push(finalTimer);
    }

    function closeBag() {
      clearBagTimers();

      bagOpen = false;
      bagState.textContent = "Packing everything back in…";

      [...bagItems].reverse().forEach((item, index) => {
        const timer = setTimeout(() => {
          item.classList.remove("out");
        }, index * 55);

        bagTimers.push(timer);
      });

      const closeTimer = setTimeout(() => {
        bagToggle.classList.remove("open");
        bagToggle.setAttribute("aria-pressed", "false");
        bagHint.textContent = "↑ tap to open";
        bagState.textContent = "Bag is closed. Go on, open it 👜";
      }, bagItems.length * 55 + 280);

      bagTimers.push(closeTimer);
    }

    function toggleBag() {
      if (bagOpen) {
        closeBag();
      } else {
        openBag();
      }
    }

    bagToggle.addEventListener("click", toggleBag);

    bagToggle.addEventListener("keydown", (event) => {
      if (event.key === "Enter" || event.key === " ") {
        event.preventDefault();
        toggleBag();
      }
    });

    const modal = document.getElementById("projectModal");

    const modalKicker = document.getElementById("modalKicker");
    const modalTitle = document.getElementById("modalTitle");
    const modalBody = document.getElementById("modalBody");

    function openProject(projectName) {
      const project = projectData[projectName];

      if (!project) return;

      modalHero.className = `modal-hero ${project.className}`;
      modalKicker.textContent = project.kicker;
      modalTitle.textContent = project.title;

      modalBody.innerHTML = `
        <p class="modal-overview">${project.overview}</p>
        ${
          project.image
            ? `<img class="modal-project-image" src="${project.image}" alt="${project.title}" onerror="this.remove()">`
            : ""
        }
        ${project.content}
      `;

      modal.classList.add("open");
      document.body.classList.add("modal-open");
      modalClose.focus();
    }

    function closeProject() {
      modal.classList.remove("open");
      document.body.classList.remove("modal-open");

      folders.forEach((folder) => {
        folder.classList.remove("open");
      });
    }

    function triggerFolder(folder) {
      folders.forEach((item) => item.classList.remove("open"));

      folder.classList.add("open");

      setTimeout(() => {
        openProject(folder.dataset.project);
      }, 550);
    }

    folders.forEach((folder) => {
      folder.addEventListener("click", () => {
        triggerFolder(folder);
      });

      folder.addEventListener("keydown", (event) => {
        if (event.key === "Enter" || event.key === " ") {
          event.preventDefault();
          triggerFolder(folder);
        }
      });
    });

    modalClose.addEventListener("click", closeProject);

    modal.addEventListener("click", (event) => {
      if (event.target === modal) {
        closeProject();
      }
    });

    document.addEventListener("keydown", (event) => {
      if (event.key === "Escape" && modal.classList.contains("open")) {
        closeProject();
      }
    });
